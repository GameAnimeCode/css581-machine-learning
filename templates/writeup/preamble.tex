% Page layout for the writeup template.
%   - First page: no header (the title block lives in the body).
%   - Page 2 onward: running header with last name left, assignment title right.
%   - Every page: footer with code repository left, "Page X of Y" right.
% Assignment macros (\asgTitle, \codeRepo, ...) are defined at the top of writeup.qmd.
% Personal macros (\stuName, \stuLast, \stuId, \stuEmail) come from personal.tex,
% which scripts/build-writeup.sh generates from .local/student.env and deletes
% afterwards. Without it, the placeholders below are used.
\InputIfFileExists{personal.tex}{}{}
\providecommand{\stuName}{First Last}
\providecommand{\stuLast}{Last}
\providecommand{\stuId}{000000000}
\providecommand{\stuEmail}{netid@uw.edu}

\usepackage{fancyhdr}
\usepackage{lastpage}

\setlength{\headheight}{14.5pt}
\renewcommand{\headrulewidth}{0pt}

\pagestyle{fancy}
\fancyhf{}
\lhead{\stuLast}
\rhead{\asgTitle}
\lfoot{\footnotesize Code Repository: \codeRepo}
\rfoot{\footnotesize Page \thepage{} of \pageref*{LastPage}}

\fancypagestyle{firstpage}{%
  \fancyhf{}%
  \renewcommand{\headrulewidth}{0pt}%
  \lfoot{\footnotesize Code Repository: \codeRepo}%
  \rfoot{\footnotesize Page \thepage{} of \pageref*{LastPage}}%
}
